\songtitle{Ging Gang Goolie}

\textit{This song was supposedly written by Scouting's founder, Robert Baden-Powell, at the first World Scout Jamboree in 1920. The song's lyrics are gibberish to allow Scouts to learn and sing the song without any language barrier.}

\begin{verse}
Ging gang goolie goolie goolie goolie watcha, \\

Ging gang goo, ging gang goo. \\

Ging gang goolie goolie goolie goolie watcha, \\

Ging gang goo, ging gang goo. \\

Hayla, oh hayla shayla, hayla shayla, shayla, oh-ho, \\

Hayla, oh hayla shayla, hayla shayla, shayla, oh. \\

Shally wally, shally wally, shally wally, shally wally, \\

Oompah, oompah, oompah, oompah. \\
\end{verse}
