\songtitle{The Song with No Name}

This is a repeat after me song \\
This is a do as I do song

\begin{verse}
A-O-A-O-A \\
\textit{*pump fists in alternating manner with the rhythm of the letters} \\
\end{verse}

\begin{verse}
A-Copesha \\
\textit{*hold fist up for the long A, and punch out arms with the following syllables} \\
\end{verse}

\begin{verse}
Copesha Longah \\
\textit{*stretch arms out from middle to as far as possible} \\
\end{verse}

\begin{verse}
Longah Choomonga \\
\textit{*stretch arms again and bounce them up and down for the Choomonga part} \\
\end{verse}

\textbf{Now sing this to your right foot}

\begin{verse}
A-O-A-O-A \\

A-Copesha \\

Copesha Longah \\

Longah Choomonga \\
\end{verse}

\textbf{Now sing this to your Scout Leader} \\
\textit{*wait a moment for Scouts to encircle their leader}

\begin{verse}
A-O-A-O-A \\

A-Copesha \\

Copesha Longah \\

Longah Choomonga \\
\end{verse}

\textbf{Now sing this to (Staff Member)} \\
\textit{*Everyone runs to the staff member}

\begin{verse}
A-O-A-O-A \\

A-Copesha \\

Copesha Longah \\

Longah Choomonga \\
\end{verse}
